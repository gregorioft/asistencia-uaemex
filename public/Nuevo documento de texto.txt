<script>

async function cargar(){

const res = await fetch("/api/admin/asistencias");
const data = await res.json();

const tabla = document.getElementById("tabla");

tabla.innerHTML="";

data.forEach(a => {

tabla.innerHTML += `
<tr>
<td>${a.numeroCuenta}</td>
<td>${a.horaEntrada || "-"}</td>
<td>${a.horaSalida || "-"}</td>
<td>${new Date(a.fecha).toLocaleDateString()}</td>
</tr>
`;

});

}

cargar();

</script>
